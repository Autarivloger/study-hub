/* Monthly habit tracker: one additive store, independent records for each month. */
let todoMode = "habits";
function habitValidMonth(value){ return typeof value==="string" && /^(?!0000)\d{4}-(0[1-9]|1[0-2])$/.test(value); }
function habitCurrentMonth(){ return todayStr().slice(0,7); }
function habitMonth(){
  const pref=DB.habitTracker.selectedMonth;
  return pref && habitValidMonth(pref.value) ? pref.value : habitCurrentMonth();
}
function habitDays(month){
  const [year,number]=month.split("-").map(Number);
  if(number===2) return year%4===0 && (year%100!==0 || year%400===0) ? 29 : 28;
  return [4,6,9,11].includes(number) ? 30 : 31;
}
function habitMonthName(month){
  const names=["January","February","March","April","May","June","July","August","September","October","November","December"];
  return names[Number(month.slice(5))-1]+" "+Number(month.slice(0,4));
}
function habitRecordKey(month,id){ return month+"|"+id; }
function habitDayKey(month,id,day){ return habitRecordKey(month,id)+"|"+String(day).padStart(2,"0"); }
function habitValidItem(key,item){
  return item && habitValidMonth(item.month) && typeof item.id==="string" && /^[a-zA-Z0-9_-]{1,100}$/.test(item.id) &&
    key===habitRecordKey(item.month,item.id) && typeof item.title==="string" && typeof item.deleted==="boolean";
}
function habitValidCheck(key,item){
  const parts=key.split("|");
  return parts.length===3 && habitValidMonth(parts[0]) && /^[a-zA-Z0-9_-]{1,100}$/.test(parts[1]) &&
    /^(0[1-9]|[12]\d|3[01])$/.test(parts[2]) && Number(parts[2])<=habitDays(parts[0]) && item && typeof item.done==="boolean";
}
function habitMerge(incoming){
  const target=DB.habitTracker;
  const time=item=>Number.isFinite(Date.parse(item && item.updatedAt)) ? Date.parse(item.updatedAt) : 0;
  const valid={months:(k,v)=>habitValidMonth(k) && v && typeof v==="object",items:habitValidItem,checks:habitValidCheck};
  Object.keys(valid).forEach(function(map){
    Object.entries(incoming[map] || {}).forEach(function([key,item]){
      if(!valid[map](key,item)) return;
      const mine=target[map][key];
      // False checks and deleted rows are retained, so old backups cannot resurrect them.
      if(!Object.prototype.hasOwnProperty.call(target[map],key) || time(item)>time(mine)) target[map][key]=item;
    });
  });
  const pref=incoming.selectedMonth;
  if(pref && habitValidMonth(pref.value) && (!target.selectedMonth || time(pref)>time(target.selectedMonth))) target.selectedMonth=pref;
}
function habitRows(month,removed){
  return Object.entries(DB.habitTracker.items).filter(([key,item])=>habitValidItem(key,item) && item.month===month && !!item.deleted===!!removed)
    .map(([,item])=>item).sort((a,b)=>(Number.isFinite(a.order)?a.order:Number.MAX_SAFE_INTEGER)-(Number.isFinite(b.order)?b.order:Number.MAX_SAFE_INTEGER) || String(a.createdAt||"").localeCompare(String(b.createdAt||"")) || a.id.localeCompare(b.id));
}
// Fill missing rows only. Names edited in a month, removed rows and daily checks stay intact.
function habitFillMonths(){
  const store=DB.habitTracker,latest=new Map();
  Object.entries(store.items).forEach(([key,item])=>{
    if(!habitValidItem(key,item)) return;
    const previous=latest.get(item.id);
    if(!previous || (Date.parse(item.updatedAt)||0)>(Date.parse(previous.updatedAt)||0)) latest.set(item.id,item);
  });
  let changed=false;
  (DB.todos||[]).forEach(task=>{
    if(!task || typeof task.title!=="string" || !task.title.trim() || !/^[a-zA-Z0-9_-]{1,90}$/.test(String(task.id))) return;
    const linked=[...latest.values()].find(item=>item.sourceTodoId===String(task.id) || item.id==='task_'+task.id);
    if(linked) return;
    const same=[...latest.values()].find(item=>item.title.trim().toLocaleLowerCase()===task.title.trim().toLocaleLowerCase());
    if(same){
      // Link an existing row instead of duplicating a task that was already added as a habit.
      Object.values(store.items).filter(item=>item.id===same.id).forEach(item=>{item.sourceTodoId=String(task.id);item.updatedAt=nowISO();});
      changed=true;return;
    }
    const at=nowISO();
    latest.set('task_'+task.id,{id:'task_'+task.id,sourceTodoId:String(task.id),title:task.title.trim(),category:typeof task.subject==="string"?task.subject:"",goalMode:"daily",goalDays:31,deleted:false,createdAt:at,updatedAt:at});
  });
  const active=[...latest.values()].filter(item=>!item.deleted);
  if(!active.length){if(changed)save();return;}
  const months=new Set(Object.keys(store.months).filter(habitValidMonth));
  Object.values(store.items).forEach(item=>{if(habitValidMonth(item.month))months.add(item.month);});
  months.add(habitMonth());
  months.forEach(month=>{
    if(!store.months[month]){store.months[month]={updatedAt:nowISO()};changed=true;}
    const rows=habitRows(month);
    let order=rows.reduce((max,item)=>Math.max(max,Number.isFinite(item.order)?item.order:-1),-1)+1;
    rows.filter(item=>!Number.isFinite(item.order)).forEach(item=>{item.order=order++;changed=true;});
    active.forEach(item=>{
      const key=habitRecordKey(month,item.id);
      if(Object.prototype.hasOwnProperty.call(store.items,key))return;
      const at=nowISO();
      store.items[key]={...item,month,order:order++,createdAt:at,updatedAt:at};changed=true;
    });
  });
  if(changed)save();
}
function habitSaveOrder(month,ids){
  const rows=habitRows(month),known=new Set(rows.map(item=>item.id));
  if(ids.length!==rows.length || new Set(ids).size!==rows.length || ids.some(id=>!known.has(id)))return;
  const at=nowISO();
  ids.forEach((id,order)=>{const item=DB.habitTracker.items[habitRecordKey(month,id)];item.order=order;item.updatedAt=at;});
  save();renderTodos();
}
function habitBindDrag(pane,month){
  pane.querySelectorAll('[data-habit-drag]').forEach(handle=>{
    handle.onkeydown=e=>{
      if(!['ArrowUp','ArrowDown'].includes(e.key))return;
      e.preventDefault();
      const ids=habitRows(month).map(item=>item.id),from=ids.indexOf(handle.dataset.habitDrag),to=from+(e.key==='ArrowUp'?-1:1);
      if(to<0 || to>=ids.length)return;
      [ids[from],ids[to]]=[ids[to],ids[from]];habitSaveOrder(month,ids);
      pane.querySelector('[data-habit-drag="'+handle.dataset.habitDrag+'"]').focus({preventScroll:true});
    };
    handle.onpointerdown=e=>{
      if(e.button!==0 || !e.isPrimary)return;
      e.preventDefault();
      const row=handle.closest('[data-habit-row]'),body=row.parentElement,grid=pane.querySelector('.ht-table-scroll'),startY=e.clientY;
      let moved=false;
      grid.setPointerCapture(e.pointerId);
      grid.onpointermove=event=>{
        if(Math.abs(event.clientY-startY)<6&&!moved)return;
        moved=true;row.classList.add('ht-dragging');
        const bounds=grid.getBoundingClientRect();
        if(event.clientY<bounds.top+35)grid.scrollTop-=18;
        if(event.clientY>bounds.bottom-35)grid.scrollTop+=18;
        const target=document.elementFromPoint(Math.max(bounds.left+8,Math.min(event.clientX,bounds.right-8)),event.clientY)?.closest('[data-habit-row]');
        if(target && target!==row && target.parentElement===body){
          const rect=target.getBoundingClientRect();body.insertBefore(row,event.clientY<rect.top+rect.height/2?target:target.nextSibling);
        }
      };
      const finish=event=>{
        grid.onpointermove=null;grid.onpointerup=null;grid.onpointercancel=null;grid.onlostpointercapture=null;
        if(grid.hasPointerCapture(e.pointerId))grid.releasePointerCapture(e.pointerId);
        row.classList.remove('ht-dragging');
        if(event.type==='pointerup' && moved)habitSaveOrder(month,[...body.querySelectorAll('[data-habit-row]')].map(item=>item.dataset.habitRow));
        else if(moved)renderTodos();
      };
      grid.onpointerup=finish;grid.onpointercancel=finish;grid.onlostpointercapture=finish;
    };
  });
}
function habitGoal(item,month){
  const days=habitDays(month),value=Number(item.goalDays);
  return item.goalMode==="daily" ? days : Math.max(1,Math.min(days,Number.isFinite(value)?Math.floor(value):days));
}
function habitCount(item,month){
  let count=0;
  for(let day=1;day<=habitDays(month);day++) if(DB.habitTracker.checks[habitDayKey(month,item.id,day)]?.done===true) count++;
  return count;
}
function habitStats(month){
  const rows=habitRows(month),daily=Array.from({length:habitDays(month)},()=>0);
  let total=0,goal=0,earned=0;
  rows.forEach(function(item){
    const count=habitCount(item,month),target=habitGoal(item,month);
    total+=count;goal+=target;earned+=Math.min(count,target);
    daily.forEach((_,i)=>{ if(DB.habitTracker.checks[habitDayKey(month,item.id,i+1)]?.done===true) daily[i]++; });
  });
  return {rows,daily,total,goal,earned,percent:goal ? Math.round(earned/goal*100) : 0};
}
function habitCopy(source,target){
  let copied=0;
  const at=nowISO();
  habitRows(source).forEach(function(item){
    const key=habitRecordKey(target,item.id);
    if(Object.prototype.hasOwnProperty.call(DB.habitTracker.items,key)) return;
    DB.habitTracker.items[key]={...item,month:target,deleted:false,createdAt:at,updatedAt:at};
    copied++;
  });
  return copied;
}
function habitEnsureMonth(month,copy){
  if(!Object.prototype.hasOwnProperty.call(DB.habitTracker.months,month)){
    const saved=Object.keys(DB.habitTracker.months).filter(habitValidMonth).sort();
    const previous=saved.filter(m=>m<month).pop() || saved[0];
    DB.habitTracker.months[month]={updatedAt:nowISO()};
    if(copy && previous) habitCopy(previous,month);
  }
}
function habitSelectMonth(month){
  if(!habitValidMonth(month)){toast("Choose a valid month.");renderTodos();return;}
  habitEnsureMonth(month,true);
  DB.habitTracker.selectedMonth={value:month,updatedAt:nowISO()};
  save();renderTodos();
}
function habitStepMonth(step){
  const [year,number]=habitMonth().split("-").map(Number),index=year*12+number-1+step;
  const y=Math.floor(index/12),m=index%12+1;
  if(y<1 || y>9999) return;
  habitSelectMonth(String(y).padStart(4,"0")+"-"+String(m).padStart(2,"0"));
}
function habitChart(stats,month){
  const width=310,height=145,top=10,left=27,bottom=25,plotHeight=height-top-bottom,plotWidth=width-left-5;
  const size=plotWidth/stats.daily.length;
  const bars=stats.daily.map(function(count,index){
    const pct=stats.rows.length ? count/stats.rows.length*100 : 0,h=pct/100*plotHeight;
    return '<rect class="ht-bar" x="'+(left+index*size+1).toFixed(2)+'" y="'+(top+plotHeight-h).toFixed(2)+'" width="'+Math.max(2,size-2).toFixed(2)+'" height="'+h.toFixed(2)+'"><title>Day '+(index+1)+': '+count+' of '+stats.rows.length+' habits ('+Math.round(pct)+'%)</title></rect>';
  }).join('');
  return '<svg class="ht-chart" viewBox="0 0 '+width+' '+height+'" role="img" aria-label="Daily habit completion for '+escapeAttr(habitMonthName(month))+'"><title>Daily completion</title><desc>'+escapeHtml(stats.daily.map((n,i)=>'Day '+(i+1)+': '+n+' of '+stats.rows.length).join('; '))+'</desc>' +
    [0,50,100].map(function(pct){const y=top+plotHeight-pct/100*plotHeight;return '<line class="ht-gridline" x1="'+left+'" y1="'+y+'" x2="'+width+'" y2="'+y+'"/><text x="1" y="'+(y+4)+'">'+pct+'%</text>';}).join('') + bars +
    '<text x="'+left+'" y="'+(height-6)+'">1</text><text x="'+(left+plotWidth/2-10)+'" y="'+(height-6)+'">Day</text><text x="'+(width-19)+'" y="'+(height-6)+'">'+stats.daily.length+'</text></svg>';
}

function habitEdit(id){
  const month=habitMonth(),days=habitDays(month),existing=id && DB.habitTracker.items[habitRecordKey(month,id)];
  openModal('<h2>'+(existing?'Edit habit':'Add habit')+'</h2><p class="ht-muted">For '+habitMonthName(month)+'</p><form id="habitForm"><div class="form-row"><label for="habitTitle">Habit name</label><input id="habitTitle" required maxlength="120" placeholder="e.g. Read for 20 minutes" value="'+escapeAttr(existing?existing.title:'')+'"></div><div class="form-row"><label for="habitCategory">Category (optional)</label><input id="habitCategory" maxlength="60" placeholder="e.g. Learning" value="'+escapeAttr(existing?existing.category||'':'')+'"></div><div class="form-row"><label for="habitGoal">Monthly goal (days)</label><input type="number" id="habitGoal" min="1" max="'+days+'" required value="'+(existing?habitGoal(existing,month):days)+'"><small>Use '+days+' for every day. You can choose a smaller goal.</small></div><p class="ht-form-error" id="habitFormError" role="alert"></p><div class="modal-actions"><button type="button" class="btn secondary" id="habitCancel">Cancel</button><button class="btn" type="submit">'+(existing?'Save changes':'Add habit')+'</button></div></form>');
  document.getElementById("habitFormError").insertAdjacentHTML('beforebegin','<div class="form-row"><label for="habitEditScope">Apply changes to</label><select id="habitEditScope"><option value="month">This month only</option><option value="all">All saved months and future months</option></select><small>New habits appear in all months. Daily ticks always stay in their own month.</small></div>');
  if(!existing){document.getElementById("habitEditScope").value='all';document.getElementById("habitEditScope").disabled=true;}
  document.getElementById("habitCancel").onclick=closeModal;
  document.getElementById("habitForm").onsubmit=e=>{
    e.preventDefault();
    const title=document.getElementById("habitTitle").value.trim(),category=document.getElementById("habitCategory").value.trim(),goal=Number(document.getElementById("habitGoal").value);
    if(!title || !Number.isInteger(goal) || goal<1 || goal>days){document.getElementById("habitFormError").textContent="Enter a habit name and a valid goal for this month.";return;}
    habitEnsureMonth(month,false);
    const at=nowISO(),item=existing || {id:uid(),month,createdAt:at,deleted:false};
    const unchangedGoal=existing && goal===habitGoal(existing,month);
    Object.assign(item,{title,category,updatedAt:at});
    if(!unchangedGoal) Object.assign(item,{goalDays:goal,goalMode:goal===days?'daily':'custom'});
    DB.habitTracker.items[habitRecordKey(month,item.id)]=item;
    if(existing && document.getElementById("habitEditScope").value==='all'){
      Object.values(DB.habitTracker.items).filter(other=>other.id===item.id).forEach(other=>{
        Object.assign(other,{title,category,goalDays:item.goalDays,goalMode:item.goalMode,updatedAt:at});
      });
    }
    DB.habitTracker.selectedMonth={value:month,updatedAt:at};
    save();closeModal();renderTodos();
  };
  document.getElementById("habitTitle").focus();
}
function renderTodos(){
  renderHabitTracker();renderTaskList();
  document.getElementById("habitPane").hidden=todoMode!=="habits";
  document.getElementById("taskPane").hidden=todoMode!=="tasks";
  document.getElementById("newTodoBtn").hidden=todoMode!=="tasks";
  document.querySelectorAll("[data-todo-mode]").forEach(function(button){
    button.setAttribute("aria-pressed",String(button.dataset.todoMode===todoMode));
    button.onclick=()=>{todoMode=button.dataset.todoMode;renderTodos();};
  });
}
